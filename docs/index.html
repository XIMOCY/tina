30秒上手

从调用大模型到构建完整 Agent，三步搞定

1. 调用大模型 
from tina.llm import BaseAPI

llm = BaseAPI()
result = llm.predict(
    input_text="Hello tina!"
)
print(result["content"])
2. 注册工具 
from tina import Tools

tools = Tools()

@tools.register()
def get_weather(city: str):
    """
    获取城市天气
    Args:
        city: 城市名称
    """
    return f"{city}今天晴，25度"
3. 创建 Agent 
from tina import Agent

agent = Agent(
    llm=BaseAPI(),
    tools=tools
)

# Agent 自动调用工具
for chunk in agent.predict("北京天气?"):
    print(chunk.get("content", ""), end="")
4. 使用 MCP 🔥 
from tina.mcp import MCPClient

mcp = MCPClient()
mcp.add_server("playwright", {
    "type": "stdio",
    "command": "npx",
    "args": ["@playwright/mcp@latest"]
})

# Agent 获得浏览器自动化能力!
agent = Agent(llm, tools, mcp=mcp)
