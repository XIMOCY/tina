### 当你进入了复杂的Agent开发，怎么更好的使用tina
如果你进入了复杂的开发的情况，例如需要有很多的来自不同功能的工具，需要有多步规划，甚至是多Agent，下面是推荐的方式和一些示例
#### 多类型工具区分
你需要使用多种不同功能的工具，推荐是按照功能，使用在类里面定义工具的方式，并在里面设定好工具包的名称防止工具冲突

#### 设定自己的Agent，使用队列来防止Agent并行
下面是一个最简单的示例
```python
from asyncio import Queue
class ChatAgent:
    def __init__(self):
        self.llm = BaseAPI()
        self.tools = Tools(name="Chat")
        self.system_prompt = "你是负责聊天的AGent"
        self.agent_core = Agent(llm=self.llm,tools=self.tools,system_prompt=self.system_prompt)

        self.message = Queue(maxsize=10)
        self.stream_chunk_processer = None

        self.tools.register_tool(tool=self.put_instruction) # 其他Agent可以给这个Agent布置任务

        

    async def put_instruction(self,instruction:str):
        await self.message.put(item=instruction)

    def set_stream_chunk_processer(self,func:callable):
        self.stream_chunk_processer = func
        self.agent_core.add_on_stream_chunk_handler(self.stream_chunk_processer) # 可以使用这个事件来把Agent的流式输出暴露出去,tina会在每一个chunk生成的时候把数据通过这个函数发布出去，支持多个处理器

    async def run(self):
        while True:
            instruction = await self.message.get()
            try:
                # 确保错误不会导致整个 Worker 挂掉
                await self.prediction(instruction)
            except Exception as e:
                print(f"Agent 执行出错: {e}")
            finally:
                self.message.task_done()

    async def prediction(self, instruction):
        async for _ in self.agent_core.apredict(instruction=instruction):
            continue

```
#### 灵活使用tina的事件系统
这里我拿一个场景举例子，假设你有一个出题的Agent，它使用工具来结构化输出，但是你发现有的时候它调用这个工具怎么办  
一般情况下，它会遵循sysprompt的设计，但是这种情况需要我们在代码层面做一个设计
```python
from asyncio import Queue
class ChatAgent:
    def __init__(self):
        self.llm = BaseAPI()
        self.tools = Tools(name="Chat")
        self.system_prompt = "你是负责聊天的AGent"
        self.agent_core = Agent(llm=self.llm,tools=self.tools,system_prompt=self.system_prompt)

        self.message = Queue(maxsize=10)
        self.stream_chunk_processer = None

        self.tools.register_tool(tool=self.put_instruction) # 其他Agent可以给这个Agent布置任务

        self.agent_core.add_on_turn_end_handler(func=self.check_some_tool_is_done) # 注意这行，它会在每次推理完结后执行，不需要参数

        

    async def put_instruction(self,instruction:str):
        await self.message.put(item=instruction)
    
    async def check_some_tool_is_done(self):
        # 在这里进行一个检查，如果没有使用对应的工具结束，就让它再试一次
        calls = self.agent_core.get_tools_call()
        if not calls or calls[-1]['name'] != "some_tool":
            await self.put_instruction(instruction="请使用some_tool来结束你这次的指令")


    def set_stream_chunk_processer(self,func:callable):
        self.stream_chunk_processer = func
        self.agent_core.add_on_stream_chunk_handler(self.stream_chunk_processer) # 可以使用这个事件来把Agent的流式输出暴露出去,tina会在每一个chunk生成的时候把数据通过这个函数发布出去，支持多个处理器

    async def run(self):
        while True:
            instruction = await self.message.get()
            try:
                # 确保错误不会导致整个 Worker 挂掉
                await self.prediction(instruction)
            except Exception as e:
                print(f"Agent 执行出错: {e}")
            finally:
                self.message.task_done()

    async def prediction(self, instruction):
        async for _ in self.agent_core.apredict(instruction=instruction):
            continue

```
